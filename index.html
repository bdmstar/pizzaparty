function showConcert(concert) { const concertContainer = document.getElementById('concert'); concertContainer.innerHTML = ''; if (Array.isArray(concert.composers)) { concert.composers.forEach(composer => { const composerElement = document.createElement('div'); composerElement.className = 'composer'; composerElement.innerHTML = `${composer.name}`; if (Array.isArray(composer.works)) { const worksList = document.createElement('ul'); composer.works.forEach(work => { const workItem = document.createElement('li'); workItem.textContent = work.title; worksList.appendChild(workItem); }); composerElement.appendChild(worksList); } concertContainer.appendChild(composerElement); }); } else { // Handle single composer format const composerElement = document.createElement('div'); composerElement.className = 'composer'; composerElement.innerHTML = `${concert.composers.name}`; if (Array.isArray(concert.composers.works)) { const worksList = document.createElement('ul'); concert.composers.works.forEach(work => { const workItem = document.createElement('li'); workItem.textContent = work.title; worksList.appendChild(workItem); }); composerElement.appendChild(worksList); } concertContainer.appendChild(composerElement); } }
